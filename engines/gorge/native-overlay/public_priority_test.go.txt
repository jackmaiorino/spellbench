func TestSpellbenchBasicSearchAfterNonbasicLondonBottom(t *testing.T) {
	search := &cards.Card{Faces: []*cards.Face{{Name: "Public Search", Abilities: []*cards.SA{{API: "ChangeZone", Params: map[string]string{"Origin": "Library", "ChangeType": "Land.Basic"}}}}}}
	spell := &cards.Card{Faces: []*cards.Face{{Name: "Public Spell", Types: []string{"Sorcery"}}}}
	forest := &cards.Card{Faces: []*cards.Face{{Name: "Forest", Types: []string{"Basic", "Land", "Forest"}}}}
	swamp := &cards.Card{Faces: []*cards.Face{{Name: "Swamp", Types: []string{"Basic", "Land", "Swamp"}}}}
	setup := PublicGame{Names: []string{"actor", "opponent"}, Decks: [][]*cards.Card{{search, spell, forest, swamp}, {spell}}}
	for _, test := range []struct {
		name, bottomName string
		owner state.PlayerID
		anonymous, afterStart bool
		want bool
	}{
		{"owned known nonbasic", "Public Spell", 0, false, false, true},
		{"owned basic", "Forest", 0, false, false, false},
		{"owned undeclared", "Unknown", 0, false, false, false},
		{"opaque owned", "Public Spell", 0, true, false, false},
		{"opaque opponent", "Public Spell", 1, true, false, true},
		{"after play began", "Public Spell", 0, false, true, false},
	} {
		t.Run(test.name, func(t *testing.T) {
			bottom := ObservedEvent{Kind: events.MoveZone, Player: test.owner, Obj: 2, From: state.ZHand, To: state.ZLibrary, Text: "bottomed"}
			identities := []Identity{{ID: 1, Name: "Public Search", Owner: 0}, {ID: 2, Name: test.bottomName, Owner: test.owner}, {ID: 3, Name: "Swamp", Owner: 0}, {ID: 4, Name: "Forest", Owner: 0}}
			if test.anonymous {bottom.Obj = 0}
			observed := []ObservedEvent{{Kind: events.Shuffle, Player: 0}, {Kind: events.Draw, Player: 0}}
			if test.afterStart {observed = append(observed, ObservedEvent{Kind: events.TurnChange})}
			observed = append(observed, bottom)
			h := History{Actor: 0, ActorBoundaries: true, Frames: []Frame{
				{Identities: identities, Events: observed},
				{Decision: &ObservedDecision{Player: 0, Kind: decision.KChoose, Source: 1, Options: []ObservedOption{
					{Action: Action{Kind: "search", Player: 0, Obj: 3}}, {Action: Action{Kind: "search", Player: 0, Obj: 4}},
				}}},
			}}
			window, ok := spellbenchBasicSearches(setup, h)[epochKey{Player: 0, Ordinal: 0}]
			if ok != test.want || (ok && (window.through != 1 || !reflect.DeepEqual(window.names, []string{"Swamp", "Forest"}))) {
				t.Fatalf("filtered search guidance: admitted=%v through=%d names=%v", ok, window.through, window.names)
			}
		})
	}
}

func TestSpellbenchPublicScryPrefersVisibleExitAndKeepsAlternatives(t *testing.T) {
	for wanted := 0; wanted < 2; wanted++ {
		e := &rules.Engine{G: state.NewGame([]string{"actor", "opponent"})}
		d := &decision.Decision{Seq: 7, Player: 1, Kind: decision.KArrange, Min: 0, Max: 2, Restable: true}
		for i, name := range []string{"First land", "Second land"} {
			obj := e.G.AddObject(&cards.Card{Faces: []*cards.Face{{Name: name}}}, 1)
			d.Options = append(d.Options, decision.Option{Kind: "bottom", Obj: obj.ID, Index: i})
		}
		fallback := decision.Intent{Seq: d.Seq, Player: 1, Choices: []int{0, 1}}
		remaining := []ObservedEvent{{Kind: events.LibraryOrder, Player: 1},
			{Kind: events.Scry, Player: 1, Amount: 1},
			{Kind: events.Draw, Player: 1},
			{Kind: events.MoveZone, Obj: 91, From: state.ZHand, To: state.ZBattlefield}}
		introduced := []Identity{{ID: 91, Name: e.G.Obj(d.Options[wanted].Obj).Card.Faces[0].Name, Owner: 1}}
		got := spellbenchObservedPreferred(e, NewCollector(0), d, fallback, remaining, introduced)
		want := decision.Intent{Seq: d.Seq, Player: 1, Choices: []int{wanted}, Rest: []int{1 - wanted}}
		if !reflect.DeepEqual(got, want) || d.Validate(got) != nil {
			t.Fatalf("public scry partition not preferred: got=%+v want=%+v", got, want)
		}
		visited, retainedFallback := 0, false
		spellbenchEachIntent(d, got, func() bool { return true }, func(in decision.Intent) bool {
			if d.Validate(in) != nil {
				t.Fatal("preferred partition admitted an illegal intent")
			}
			if visited == 0 && !reflect.DeepEqual(in, want) {
				t.Fatal("preferred partition was not visited first")
			}
			retainedFallback = retainedFallback || reflect.DeepEqual(in, fallback)
			visited++
			return true
		})
		if visited < 6 || !retainedFallback {
			t.Fatalf("legal alternatives were pruned: count=%d fallback=%v", visited, retainedFallback)
		}
	}
}

func TestSpellbenchPublicScryRequiresMatchingPublicMarker(t *testing.T) {
	e := &rules.Engine{G: state.NewGame([]string{"actor", "opponent"})}
	d := &decision.Decision{Seq: 8, Player: 1, Kind: decision.KArrange, Min: 0, Max: 2, Restable: true,
		Options: []decision.Option{{Kind: "bottom", Index: 0}, {Kind: "bottom", Index: 1}}}
	fallback := decision.Intent{Seq: d.Seq, Player: 1, Choices: []int{0, 1}}
	for _, remaining := range [][]ObservedEvent{
		nil,
		{{Kind: events.Scry, Player: 1, Amount: 1}},
		{{Kind: events.LibraryOrder, Player: 0}, {Kind: events.Scry, Player: 1, Amount: 1}},
		{{Kind: events.LibraryOrder, Player: 1}, {Kind: events.Scry, Player: 0, Amount: 1}},

	} {
		got := spellbenchObservedPreferred(e, NewCollector(0), d, fallback, remaining, nil)
		if !reflect.DeepEqual(got, fallback) {
			t.Fatalf("unmatched marker changed arrangement: events=%+v got=%+v", remaining, got)
		}
	}
}

func TestSpellbenchPublicScryHeldProposalPreservesOriginalConstraints(t *testing.T) {
	key := epochKey{Player: 1}
	original := map[epochKey]epochConstraints{key: {Deadlines: []deadlineConstraint{
		{Through: 1, Name: "Already consumed", Count: 1},
		{Through: 4, Name: "Later public card", Count: 1},
	}}}
	for _, player := range []state.PlayerID{0, 1} {
		h := History{Actor: 0, Frames: []Frame{{Events: []ObservedEvent{
			{Kind: events.Shuffle, Player: 1}, {Kind: events.Draw, Player: 1}, {Kind: events.Draw, Player: 1},
			{Kind: events.LibraryOrder, Player: player}, {Kind: events.Draw, Player: 1},
			{Kind: events.LibraryOrder, Player: player},
		}}}}
		guided := spellbenchScryHeldEpochs(h, original)
		if len(original[key].Deadlines) != 2 {
			t.Fatal("optional held proposal changed original native constraints")
		}
		if player == 0 {
			if !reflect.DeepEqual(guided[key], original[key]) {
				t.Fatal("actor reorder constrained the opponent")
			}
			continue
		}
		bounds := guided[key].Deadlines
		if len(bounds) != 3 || bounds[2].Name != "Later public card" || bounds[2].Through != 2 || bounds[2].Count != 1 {
			t.Fatalf("held proposal did not precede the first reorder: %+v", bounds)
		}
		// The copied prefix must have independent storage as well as values.
		bounds[0].Count = 9
		if original[key].Deadlines[0].Count != 1 {
			t.Fatal("held witness aliased original deadline storage")
		}
	}
}

func TestSpellbenchPublicPassBeforeLaterAction(t *testing.T) {
	e := &rules.Engine{G: state.NewGame([]string{"p0", "p1", "p2"})}
	e.G.Players[2].Lost = true
	d := &decision.Decision{Seq: 7, Player: 1, Kind: decision.KPriority, Min: 1, Max: 1,
		Options: []decision.Option{{Kind: "concede"}, {Kind: "pass"}}}
	fallback := decision.Intent{Seq: d.Seq, Player: d.Player, Choices: []int{0}}
	want := decision.Intent{Seq: d.Seq, Player: d.Player, Choices: []int{1}}
	// The saved failure's public burst starts with the opponent's pass, then
	// includes later actions. Those later actions cannot precede the marker.
	remaining := []ObservedEvent{{Kind: events.Priority, Player: 0, Amount: 1},
		{Kind: events.PutOnStack, Player: 1, Obj: 101, From: state.ZHand}}
	got := spellbenchObservedPreferred(e, NewCollector(0), d, fallback, remaining, nil)
	if !reflect.DeepEqual(got, want) || d.Validate(got) != nil {
		t.Fatalf("public pass was not preferred: got=%+v want=%+v", got, want)
	}
	// Ordering must retain the alternative when the preferred branch fails.
	var visited []decision.Intent
	spellbenchEachIntent(d, got, func() bool { return true }, func(in decision.Intent) bool {
		visited = append(visited, in)
		return true
	})
	if len(visited) != 2 || !reflect.DeepEqual(visited[0], want) || !reflect.DeepEqual(visited[1], fallback) {
		t.Fatalf("legal alternatives changed: %+v", visited)
	}
}

func TestSpellbenchPublicPassRequiresImmediateMatchingMarker(t *testing.T) {
	e := &rules.Engine{G: state.NewGame([]string{"p0", "p1"})}
	d := &decision.Decision{Seq: 9, Player: 1, Kind: decision.KPriority, Min: 1, Max: 1,
		Options: []decision.Option{{Kind: "concede"}, {Kind: "pass"}}}
	fallback := decision.Intent{Seq: d.Seq, Player: d.Player, Choices: []int{0}}
	for _, remaining := range [][]ObservedEvent{
		nil,
		{{Kind: events.Priority, Player: 0, Amount: 0}},
		{{Kind: events.Priority, Player: 1, Amount: 1}},
		{{Kind: events.ManaClear, Player: 1}, {Kind: events.Priority, Player: 0, Amount: 1}},
	} {
		got := spellbenchObservedPreferred(e, NewCollector(0), d, fallback, remaining, nil)
		if !reflect.DeepEqual(got, fallback) {
			t.Fatalf("unmatched marker changed preference: events=%+v got=%+v", remaining, got)
		}
	}
	// A non-priority question must keep its own legal answer.
	d.Kind = decision.KChoose
	got := spellbenchObservedPreferred(e, NewCollector(0), d, fallback,
		[]ObservedEvent{{Kind: events.Priority, Player: 0, Amount: 1}}, nil)
	if !reflect.DeepEqual(got, fallback) {
		t.Fatalf("priority marker changed another question: %+v", got)
	}
}

func TestSpellbenchPublicCarryCreditsReturnedCards(t *testing.T) {
	for _, test := range []struct {
		name string
		from state.Zone
		exit bool
		blind bool
		wantLandscape int
	}{
		{"graveyard return remains held", state.ZGraveyard, false, false, 1},
		{"public exile return remains held", state.ZExile, false, false, 1},
		{"library draw supplies no return credit", state.ZLibrary, false, false, 2},
		{"renamed public exit spends credit", state.ZGraveyard, true, false, 2},
		{"opaque exit clears credit", state.ZGraveyard, true, true, 2},
	} {
		t.Run(test.name, func(t *testing.T) {
			h := History{Actor: 0, ActorBoundaries: true, Frames: []Frame{
				{Events: []ObservedEvent{{Kind: events.Shuffle, Player: 1}, {Kind: events.Draw, Player: 1}, {Kind: events.Draw, Player: 1}}},
				{Identities: []Identity{{ID: 14, Name: "Twisted Landscape", Owner: 1}},
					Events: []ObservedEvent{{Kind: events.MoveZone, Obj: 14, From: test.from, To: state.ZHand}}},
			}}
			if test.exit {
				frame := Frame{Events: []ObservedEvent{{Kind: events.MoveZone, Obj: 101, From: state.ZHand, To: state.ZBattlefield}}}
				if !test.blind {
					frame.Identities = []Identity{{ID: 101, Name: "Twisted Landscape", Owner: 1}}
				}
				h.Frames = append(h.Frames, frame)
			}
			h.Frames = append(h.Frames, Frame{Board: []byte(`{"players":[{"seat":1,"hand_size":2}]}`),
				Events: []ObservedEvent{{Kind: events.TurnChange, Player: 1}, {Kind: events.Shuffle, Player: 1}}})
			previous, next := epochKey{Player: 1}, epochKey{Player: 1, Ordinal: 1}
			original := map[epochKey]epochConstraints{
				previous: {Deadlines: []deadlineConstraint{{Through: 1, Name: "Twisted Landscape", Count: 1}}},
				next: {Deadlines: []deadlineConstraint{{Through: 0, Name: "Twisted Landscape", Count: 1}, {Through: 0, Name: "Refurbished Familiar", Count: 1}}},
			}
			got := spellbenchCarryEpochs(h, original)
			landscape, familiar := 0, 0
			for _, bound := range got[previous].Deadlines {
				if bound.Name == "Twisted Landscape" {
					landscape = max(landscape, bound.Count)
				}
				if bound.Name == "Refurbished Familiar" {
					familiar = max(familiar, bound.Count)
				}
			}
			if landscape != test.wantLandscape || familiar != 1 {
				t.Fatalf("return carry requested an extra drawn copy: landscape=%d familiar=%d deadlines=%+v", landscape, familiar, got[previous].Deadlines)
			}
			if len(original[previous].Deadlines) != 1 || !reflect.DeepEqual(got[next], original[next]) {
				t.Fatal("witness carry changed original or later native constraints")
			}
		})
	}
}

func TestSpellbenchPublicCarrySurvivesScryBottom(t *testing.T) {
	previous, next := epochKey{Player: 1}, epochKey{Player: 1, Ordinal: 1}
	for _, test := range []struct {
		name string
		reorderPlayer state.PlayerID
		reorders bool
		wantPrefix int
	}{
		{"before opponent reorder", 1, true, 2},
		{"actor reorder does not constrain opponent", 0, true, 4},
		{"no reorder", 1, false, 4},
	} {
		t.Run(test.name, func(t *testing.T) {
			h := History{Actor: 0, ActorBoundaries: true, Frames: []Frame{
				{Events: []ObservedEvent{{Kind: events.Shuffle, Player: 1}, {Kind: events.Draw, Player: 1}, {Kind: events.Draw, Player: 1}}},
			}}
			if test.reorders {
				h.Frames = append(h.Frames, Frame{Events: []ObservedEvent{{Kind: events.LibraryOrder, Player: test.reorderPlayer}}})
			}
			h.Frames = append(h.Frames, Frame{Events: []ObservedEvent{{Kind: events.Draw, Player: 1}}})
			if test.reorders {
				h.Frames = append(h.Frames, Frame{Events: []ObservedEvent{{Kind: events.LibraryOrder, Player: test.reorderPlayer}}})
			}
			h.Frames = append(h.Frames, Frame{Events: []ObservedEvent{{Kind: events.Draw, Player: 1}}},
				Frame{Board: []byte(`{"players":[{"seat":1,"hand_size":2}]}`),
					Events: []ObservedEvent{{Kind: events.TurnChange, Player: 1}, {Kind: events.Shuffle, Player: 1}}})
			// One copy is consumed after the first reorder. Another must still
			// be held at the next shuffle, so both must precede the reorder.
			original := map[epochKey]epochConstraints{
				previous: {Deadlines: []deadlineConstraint{{Through: 3, Name: "Refurbished Familiar", Count: 1}}},
				next: {Deadlines: []deadlineConstraint{{Through: 0, Name: "Refurbished Familiar", Count: 1}}},
			}
			guided := spellbenchCarryEpochs(h, original)
			bounds := guided[previous].Deadlines
			if len(bounds) != 2 || bounds[1].Through != test.wantPrefix || bounds[1].Count != 2 {
				t.Fatalf("carry card could remain in the reordered tail: %+v", bounds)
			}
			if !reflect.DeepEqual(guided[next], original[next]) || len(original[previous].Deadlines) != 1 {
				t.Fatal("optional carry changed native constraints")
			}
			if !test.reorders || test.reorderPlayer != 1 {
				return
			}
			library := []rules.ShuffleCard{{ID: 1, Name: "Refurbished Familiar"}, {ID: 2, Name: "Refurbished Familiar"},
				{ID: 3, Name: "Other"}, {ID: 4, Name: "Other"}, {ID: 5, Name: "Other"}, {ID: 6, Name: "Other"}}
			for attempt := 0; attempt < 8; attempt++ {
				p := &proposalState{epochs: guided, base: 27, attempt: attempt, observer: NewCollector(0),
					result: &SampleResult{}, plans: newConstraintPlanCache(), exclusions: newExclusionStore()}
				order, err := p.plan(rules.ShuffleContext{Player: 1, Ordinal: 0, Library: library})
				if err != nil || len(order) != len(library) {
					t.Fatalf("carry shuffle failed: order=%v err=%v", order, err)
				}
				// Draw two, bottom the next card, then draw its successor. The
				// required copies must already be held before the scry.
				copies := 0
				for _, id := range order[:2] {
					if id == 1 || id == 2 {
						copies++
					}
				}
				if copies != 2 {
					t.Fatalf("scry can bottom a required carry copy: order=%v", order)
				}
			}
		})
	}
}
