package searchprobe

import (
	"bytes"
	"encoding/json"
	"io"
	"reflect"
)

// Spellbench's reference host canonicalizes JSON object keys. RawMessage
// boards must retain every field and array position, but key order is not an
// observation. UseNumber preserves integer values without float64 rounding.
func spellbenchBoardEqual(got, want json.RawMessage, public bool) bool {
	if !public {
		return bytes.Equal(got, want)
	}
	if bytes.Equal(got, want) {
		trimmed := bytes.TrimSpace(got)
		return len(trimmed) > 0 && trimmed[0] == '{' && json.Valid(got)
	}
	decode := func(raw json.RawMessage) (map[string]any, bool) {
		decoder := json.NewDecoder(bytes.NewReader(raw))
		decoder.UseNumber()
		var board map[string]any
		if decoder.Decode(&board) != nil || board == nil {
			return nil, false
		}
		var trailing any
		if decoder.Decode(&trailing) != io.EOF {
			return nil, false
		}
		return board, true
	}
	a, ok := decode(got)
	if !ok {
		return false
	}
	b, ok := decode(want)
	return ok && reflect.DeepEqual(spellbenchWithoutResidue(a), spellbenchWithoutResidue(b))
}

// Battlefield entry always sets summoning sickness, and only an untap step on
// the battlefield clears it. Off the battlefield the bit is residue of an
// earlier visit with no rules effect, and it distinguishes otherwise
// identical copies returned to a hidden library. Ignore only that bit, and
// only outside the battlefield; every other field and position still counts.
func spellbenchWithoutResidue(board map[string]any) map[string]any {
	var strip func(value any, battlefield bool) any
	strip = func(value any, battlefield bool) any {
		switch v := value.(type) {
		case map[string]any:
			out := make(map[string]any, len(v))
			for key, item := range v {
				if key == "summon_sick" && !battlefield {
					continue
				}
				out[key] = strip(item, battlefield || key == "battlefield")
			}
			return out
		case []any:
			out := make([]any, len(v))
			for i, item := range v {
				out[i] = strip(item, battlefield)
			}
			return out
		}
		return value
	}
	return strip(board, false).(map[string]any)
}

func spellbenchFrameEqual(got, want Frame) bool {
	a, b := got.Board, want.Board
	got.Board, want.Board = nil, nil
	return reflect.DeepEqual(got, want) && spellbenchBoardEqual(a, b, true)
}
